    function showErrorToast(message) {
        // Usa uma mensagem padrão se nenhuma for fornecida
        message = message || 'Ocorreu um erro ao processar sua solicitação.';

        // Cria o HTML do toast dinamicamente
        const toastHtml = `
            <div class="toast align-items-center text-white bg-danger border-0" role="alert" aria-live="assertive" aria-atomic="true">
                <div class="d-flex">
                    <div class="toast-body">
                        <i class="fas fa-exclamation-triangle me-2"></i>
                        ${message}
                    </div>
                    <button type="button" class="btn-close btn-close-white me-2 m-auto" data-bs-dismiss="toast" aria-label="Close"></button>
                </div>
            </div>
        `;

        const $toast = $(toastHtml);
        // Adiciona o toast ao container que criamos no layout
        $('.toast-container').append($toast);

        // Inicializa e exibe o toast do Bootstrap
        const toast = new bootstrap.Toast($toast[0]);
        toast.show();

        // Opcional: Remove o toast do HTML depois que ele desaparece
        $toast.on('hidden.bs.toast', function() {
            $(this).remove();
        });
    }
    $(document).ready(function() {
        // Inicializa o Select2
        $('#campoMunicipio').select2({
            theme: 'bootstrap-5'
        });

        var tabelaResultado = $('#tblResultado').DataTable({
            paging: true,
            lengthChange: true,
            searching: true,
            ordering: true,
            info: true,
            autoWidth: false,
            responsive: true,
            language: {
                url: "https://consultas.ima.sc.gov.br/lib/dataTables/pt-br.json"
            },
            dom: 'Bfrtip',
            buttons: [{
                    extend: 'csv',
                    text: '<i class="fas fa-file-csv me-1"></i> Exportar CSV',

                },
                {
                    extend: 'print',
                    text: '<i class="fas fa-print me-1"></i> Imprimir',
                    className: 'btn btn-info btn-sm'
                }
            ],
            // --- INÍCIO DAS MUDANÇAS ---
            columns: [{
                    data: 'fce'
                },
                {
                    data: 'razaoSocial'
                }, // Corrigido para corresponder à view do BD
                {
                    data: 'cnpj'
                }, // Corrigido
                {
                    data: 'atividadeConsema'
                }, // Corrigido
                {
                    data: 'dtEntrada'
                }, // Corrigido
                {
                    data: 'faseDescricao'

                }, // Corrigido
                {
                    data: null, // Não está vinculado a uma única coluna de dados
                    orderable: false, // Não permite ordenar por esta coluna
                    className: 'text-center', // Centraliza o conteúdo
                    // A função 'render' cria o HTML do botão dinamicamente
                    render: function(data, type, row) {
                        

                        // Pega os dados necessários da linha atual
                        var numeroDoc = row.numeroDocumento;
                        var fce = row.fce;
                        var tipoProcesso = row.fce_processo_tipo;

                        // Condição: Só mostra o botão se 'numeroDocumento' não for nulo, vazio ou igual a "-"
                        if (numeroDoc && numeroDoc.trim() !== '' && numeroDoc.trim() !== '-') {

                            var url = `https://consultas.ima.sc.gov.br/consulta/visualizar/${fce}/${tipoProcesso}/${numeroDoc.replace(/\//g, '-')}`;

                            // Retorna o HTML do botão de olho
                            return `<a target="_blank" href="${url}" class="btn btn-info btn-sm" title="Visualizar Detalhes"><i class="fas fa-eye"></i></a>`;
                        }else{
                            return `-`;
                        }


                        // Se a condição não for atendida, retorna uma string vazia (nenhum botão)
                        return '';
                    }
                }
            ]

        // Evento de submit do formulário de pesquisa
        $('#form-pesquisa').on('submit', function(e) {
            
            e.preventDefault();

            
            // --- INÍCIO: VALIDAÇÃO PARA IMPEDIR BUSCA VAZIA ---
            
            var formData = $(this).serializeArray();
            var isFormEmpty = true;

            // Percorre todos os campos do formulário para verificar se estão vazios
            for (var i = 0; i < formData.length; i++) {
                // Ignora o token do reCAPTCHA na verificação
                if (formData[i].name !== 'g-recaptcha-response' && formData[i].value.trim() !== '') {
                    isFormEmpty = false;
                    break; // Encontrou um campo preenchido, pode parar de verificar
                }
            }

            // Se o formulário estiver completamente vazio, exibe um erro e para a execução da função
            if (isFormEmpty) {
                showErrorToast('Por favor, preencha pelo menos um campo para realizar a busca.');
                return; // Para a execução aqui
            }

            // --- FIM: VALIDAÇÃO ---

            var $button = $(this).find('button[type="submit"]');
            var $icon = $button.find('i');
            var originalIconClass = 'fas fa-search me-1';
            var loadingIconClass = 'fas fa-spinner fa-spin me-1';

            $button.prop('disabled', true);
            $icon.attr('class', loadingIconClass);

            $.ajax({
                url: "https://consultas.ima.sc.gov.br/consulta/consultar",
                type: 'POST',
                data: $(this).serialize(),
                dataType: 'json',
                success: function(data) {
                    tabelaResultado.clear();
                    if (data && data.length > 0) {
                        tabelaResultado.rows.add(data);
                    }
                    tabelaResultado.draw();
                },
                error: function(jqXHR, textStatus, errorThrown) {
                    console.error("Erro AJAX:", textStatus, errorThrown, jqXHR.responseText);
                    showErrorToast('Ocorreu um erro ao realizar a consulta.');
                },
                complete: function() {
                    $button.prop('disabled', false);
                    $icon.attr('class', originalIconClass);
                    grecaptcha.reset();
                    document.getElementById("botaoConsultar").disabled = true;
                }
            });
        });



    });
