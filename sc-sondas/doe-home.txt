        document.addEventListener("DOMContentLoaded", function() {
            console.log("🚀 Script iniciado!");

            var currentURL = window.location.href;
            console.log("🔍 URL Atual:", currentURL);

            var oldVersionRegex = /\/v\d{1,3}\.\d{1,2}\.\d{1,2}\//; // Formato vX.XX.XX
            var newVersion = "v2.46.00";

            var oldDomain = "https://doe.sea.sc.gov.br";
            var newDomain = "https://portal.doe.sea.sc.gov.br";
            
            var updatedURL = currentURL;

            // Se o domínio for o antigo, substituí-lo
            if (currentURL.startsWith(oldDomain)) {
                updatedURL = currentURL.replace(oldDomain, newDomain);
                console.log("🔄 Domínio atualizado:", updatedURL);
            }

            // Verifica se já contém uma versão na URL e substitui
            if (oldVersionRegex.test(updatedURL)) {
                updatedURL = updatedURL.replace(oldVersionRegex, '/' + newVersion + '/');
                console.log("📌 Versão antiga substituída:", updatedURL);
            } 
            // Se não contiver versão, insere a nova versão após o domínio
            else if (!/\/v\d+\.\d+\.\d+\//.test(updatedURL)) {
                var urlParts = updatedURL.split('/');
                urlParts.splice(3, 0, newVersion); 
                updatedURL = urlParts.join('/');
                console.log("📌 Versão adicionada:", updatedURL);
            }

            if (updatedURL !== currentURL) {
                console.log("✅ Redirecionando para:", updatedURL);
                document.getElementById("redirectMessage").innerHTML = "Redirecionando para: " + updatedURL;
                
                setTimeout(() => {
                    window.location.href = updatedURL;
                }, 2000);
            } else {
                console.warn("⚠️ Nenhuma alteração na URL detectada. Verifique a lógica.");
            }
        });
