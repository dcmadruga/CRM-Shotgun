                success: function(data) {
                    
                    tabelaResultado.clear();
                    if (data && data.length > 0) {
                        tabelaResultado.rows.add(data);
                    }
                    tabelaResultado.draw();
                },
                error: function(jqXHR, textStatus, errorThrown) {
                    console.error("Erro AJAX:", textStatus, errorThrown, jqXHR.responseText);
                    alert('Ocorreu um erro ao realizar a consulta.');
                },
                complete: function() {
                    $button.prop('disabled', false);
                    $icon.attr('class', originalIconClass);
                    // Reseta o reCAPTCHA para uma nova consulta
                    grecaptcha.reset();
                    document.getElementById("botaoConsultar").disabled = true;
                }
            });
        });

        // Lógica para desabilitar o campo de atividade
        $("#campoTipo").change(function() {
            if ($("#campoTipo").val() == "EXPLORACAO_FLORESTAL") {
                $("#campoAtividade").prop("disabled", true);
                $("#campoPorte").prop("disabled", true);
                $("#campoPotencialPoluidor").prop("disabled", true);
            } else {
                if ($("#campoTipo").val() == "CERTIDAO") {
                    $("#campoPorte").prop("disabled", true);
                } else {
                    $("#campoPorte").prop("disabled", false);
                    $("#campoAtividade").prop("disabled", false);
                    $("#campoPotencialPoluidor").prop("disabled", false);
                }

            }
        });


        // Ativa o Select2 para o campo de Atividade
        $('#campoAtividade').select2({
            theme: 'bootstrap-5',
            placeholder: 'Selecione uma atividade'
        });

        // Ativa o Select2 para o campo de Município
        $('#campoMunicipio').select2({
            theme: 'bootstrap-5',
            placeholder: 'Selecione um município'
        });

        // Ativa o Select2 para o campo de Regional
        $('#campoRegional').select2({
            theme: 'bootstrap-5',
            placeholder: 'Selecione uma regional'
        });

    });
</script>
