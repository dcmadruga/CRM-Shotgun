document.addEventListener("DOMContentLoaded", function() {
    const container = document.getElementById('mapa-sc-container');
    if (!container) return;

    fetch('/static/mapa-sc.svg')
        .then(response => response.text())
        .then(svgText => {
            container.innerHTML = svgText;

            const rawData = container.getAttribute('data-municipios-ibge');
            const ativosDoSinfat = rawData.replace(/[\[\]']/g, '').split(',').map(s => s.trim());

            ativosDoSinfat.forEach(codigo => {
                if (!codigo) return;
                pathMunicipio = container.querySelector('path[id*="' + codigo + '"]');
                if (pathMunicipio) {
                    pathMunicipio.classList.add('municipio-ativo');
                } else {
                    console.warn('Não foi possível encontrar o elemento visual para o código: ' + codigo);
                }
            });
        })
        .catch(err => console.error('Erro ao processar a renderização do mapa:', err));
});
