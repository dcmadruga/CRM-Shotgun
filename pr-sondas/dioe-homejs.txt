var EDICAO_ID      = 0;
var TIPO_ATUAL     = '';
var TIPO_EDICAO_ID = null; // ID do tipo selecionado, enviado à API ao buscar por data/número

function ultima_data() {
    $.ajax({
        url: BASE_URL + '/apifront/portal/edicoes/edicoes_from_data.json',
        data: {
            subtheme: SUBTEMA
        }
    }).done(function(data) {
        var obj = data;
        if (obj.erro) {
            alert(obj.msg);
            return;
        }
        if (obj.itens && obj.itens[0]) {
            $("#dataEdicaoPortal").val(obj.itens[0].data);
            EDICAO_ID  = obj.itens[0].id;
            TIPO_ATUAL = obj.itens[0].tipo_edicao_nome || '';
            $('.tipo-atual-nome').text(TIPO_ATUAL || 'Diário Oficial');
            carregarImagem(EDICAO_ID);
            definirUrlDiarios(EDICAO_ID);
            $('.numero-edicao').html(formatar(obj.itens[0].numero));
            $('.data-edicao').html(formatarData(obj.itens[0].data));
            $('.paginas-edicao').html(obj.itens[0].paginas ? ' · ' + obj.itens[0].paginas + ' págs.' : '');
            if (window.TEMAS_MAPA !== undefined) {
                carregarTiposHomeDinamico(window.TEMAS_MAPA);
            }
        }
    });
}

function ultimos_diarios() {
    $.ajax({
        url: BASE_URL + '/apifront/portal/edicoes/ultimas_edicoes.json',
        data: {
            subtheme: SUBTEMA
        }
    }).done(function(data) {
        var obj = data;
        if (obj.erro) {
            return;
        }
        var levls = {};
        if (obj.itens.length > 0) {
            for (var t in obj.itens) {
                if (!levls[obj.itens[t].data]) {
                    levls[obj.itens[t].data] = [];
                }
                var curr = {
                    'nome': obj.itens[t].tipo_edicao_nome,
                    'id': obj.itens[t].id
                };
                if (obj.itens[t].suplemento == 1) {
                    curr.nome = curr.nome + ' SUPLEMENTO';
                }
                levls[obj.itens[t].data].push(curr);
            }
        }
        var op = '';
        for (var t in levls) {
            if (op != '') op += '</optgroup>';
            op += '<optgroup label="' + t + '">';
            for (var g in levls[t]) {
                op += '<option value="' + levls[t][g].id + '" >' + levls[t][g].nome + '</option>';
            }
        }
        $("#selectEdicaoPortal").html(op);
    });
}

function carregarImagem(edicao_id) {
    definirUrlDiarios(edicao_id);
    if (edicao_id > 0) {
        var url = (CDN_URL != '' ? CDN_URL : BASE_URL) + ((SUBTEMA) ? '/' + SUBTEMA : '') + '/apifront/portal/edicoes/imagem_diario/' + edicao_id + '/1/imagem';
        var imagem = $('<div class="processing"></div>');
        $("#imagemCapa").parent().append(imagem);
        $("#imagemCapa").fadeOut('fast');
        var im = new Image();
        im.onload = function() {
            $("#imagemCapa").attr('src', url).fadeIn('fast');
            $("#imagemCapa").parent().find(".processing").fadeOut('slow', function() {
                $(this).remove();
            });
        }
        im.src = url;
    }
}

function definirUrlDiarios(edicao_id) {
    if (edicao_id > 0) {
        var urlPdf = BASE_URL + ((SUBTEMA) ? '/' + SUBTEMA : '') + '/portal/visualizacoes/pdf/' + edicao_id + '/#e:' + edicao_id;
        var urlHTML = BASE_URL + ((SUBTEMA) ? '/' + SUBTEMA : '') + '/portal/visualizacoes/html/' + edicao_id + '/#e:' + edicao_id;
        var urlJornal = BASE_URL + ((SUBTEMA) ? '/' + SUBTEMA : '') + '/portal/visualizacoes/jornal/' + edicao_id + '/#e:' + edicao_id;
        var urlFull = (CDN_URL != '' ? CDN_URL : BASE_URL) + '/portal/edicoes/download/' + edicao_id;
        var urlP7s = (CDN_URL != '' ? CDN_URL : BASE_URL) + '/portal/edicoes/downloadP7s/' + edicao_id;
        $("#downloadPdf").attr('href', urlPdf);
        $("#downloadPdfView").attr('href', urlPdf);
        $("#downloadHTML").attr('href', urlHTML);

        $("#baixar-diario-p7s").attr('href', urlP7s);

    }
}

function populaAbas(obj) {
    if (obj.length > 1) {
        if (obj[1].suplemento == 1) {
            $('#boxabas li:eq(0)').addClass('active').data('idDio', obj[0].id);
            $('#boxabas li:eq(1)').data('idDio', obj[1].id);
        } else {
            $('#boxabas li:eq(0)').addClass('active').data('idDio', obj[1].id);
            $('#boxabas li:eq(1)').data('idDio', obj[0].id);
        }
        $('#boxabas li.active').removeClass('active');
        $('#boxabas li:eq(0)').addClass('active');
        $('#boxabas a').unbind('click').bind('click', function(e) {
            e.preventDefault();
            $('#boxabas li.active').removeClass('active');
            $(this).parent().addClass('active');
            EDICAO_ID = $(this).parent().data('idDio');
            carregarImagem(EDICAO_ID);
        });
        $('#boxabas').slideDown('fast');
    }
}

// Estado dos tipos — compartilhado entre carregarTiposHomeDinamico, renderizarTiposGrid e selecionarTipo
var _tiposVistos   = [];
var _latestPorTipo = {};
var _nomeParaSlug  = {};

function carregarTiposHomeDinamico(temasMapa) {
    // Monta mapeamento nome → slug para os subthemes configurados
    _nomeParaSlug = {};
    Object.keys(temasMapa).forEach(function(slug) {
        var nome = (temasMapa[slug].nome || '').trim();
        if (nome) _nomeParaSlug[nome] = slug;
    });

    $.ajax({
        url: BASE_URL + '/apifront/portal/edicoes/ultimas_edicoes.json?sem_limite=1&todos=1'
    }).done(function(data) {
        if (data.erro || !data.itens || data.itens.length === 0) return;

        _tiposVistos   = [];
        _latestPorTipo = {};
        data.itens.forEach(function(ed) {
            var tipo = ed.tipo_edicao_nome || '';
            if (tipo && _tiposVistos.indexOf(tipo) === -1) {
                _tiposVistos.push(tipo);
                _latestPorTipo[tipo] = ed;
            }
        });

        // Ordena pela posição definida pelo admin
        _tiposVistos.sort(function(a, b) {
            var pa = _latestPorTipo[a].tipo_edicao_posicao || 0;
            var pb = _latestPorTipo[b].tipo_edicao_posicao || 0;
            return pa - pb;
        });

        renderizarTiposGrid();
    });
}

function renderizarTiposGrid() {
    var container = document.getElementById('tipos-grid');
    if (!container) return;

    var html = '';
    var c = 0;
    _tiposVistos.forEach(function(tipo) {
        if (c > 3) return;
        var ed     = _latestPorTipo[tipo];
        var slug   = _nomeParaSlug[tipo] || '';
        var cardId = tipo.toLowerCase().replace(/[^a-z0-9]/g, '-');
        var urlPdf = BASE_URL + (slug ? '/' + slug : '') + '/portal/visualizacoes/pdf/' + ed.id + '/#e:' + ed.id;
        var supl   = ed.suplemento == 1 ? ' <small style="color:#e8a000;font-weight:700;">SUPL</small>' : '';
        var tipoEsc = tipo.replace(/\\/g, '\\\\').replace(/'/g, "\\'");

        html += '<div class="tipo-card">';
        html +=   '<div class="tipo-card-thumb">';
        html +=     '<img id="capa-tipo-' + cardId + '" src="" alt="' + tipo + '" class="tipo-capa-img" style="display:none;">';
        html +=     '<div class="tipo-capa-loading" id="loading-tipo-' + cardId + '"></div>';
        html +=   '</div>';
        html +=   '<div class="tipo-card-body">';
        html +=     '<h3>' + tipo + '</h3>';
        html +=     '<p class="tipo-edicao-meta">Edição Nº ' + formatar(ed.numero) + supl + ' · ' + formatarData(ed.data) + '</p>';
        html +=     '<div class="tipo-card-actions">';
        html +=       '<button class="btn-acessar" onclick="selecionarTipo(\'' + tipoEsc + '\')"><svg width="14" height="14" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><circle cx="11" cy="11" r="8"/><path d="m21 21-4.35-4.35"/></svg> Acessar</button>';
        html +=       '<a href="' + urlPdf + '" class="btn-pdf-card" target="_blank"><svg width="14" height="14" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M12 15V3m0 12-4-4m4 4 4-4M2 17l.621 2.485A2 2 0 0 0 4.561 21h14.878a2 2 0 0 0 1.94-1.515L22 17"/></svg> PDF</a>';
        html +=     '</div>';
        html +=   '</div>';
        html += '</div>';

        c++;
    });
    container.innerHTML = html;

    // Carrega imagens de capa para cada tipo (exceto o atual)
    var c = 0;
    _tiposVistos.forEach(function(tipo) {
        if (c > 3) return;
        var ed     = _latestPorTipo[tipo];
        var slug   = _nomeParaSlug[tipo] || '';
        var cardId = tipo.toLowerCase().replace(/[^a-z0-9]/g, '-');
        var base   = CDN_URL !== '' ? CDN_URL : BASE_URL;
        var imgUrl = base + (slug ? '/' + slug : '') + '/apifront/portal/edicoes/imagem_diario/' + ed.id + '/1/imagem';
        var $img   = $('#capa-tipo-' + cardId);
        var $load  = $('#loading-tipo-' + cardId);
        var im = new Image();
        im.onload  = function() { $img.attr('src', imgUrl).fadeIn('fast'); $load.hide(); };
        im.onerror = function() { $load.hide(); };
        im.src = imgUrl;
        c++;
    });
}

function selecionarTipo(tipo) {
    var ed = _latestPorTipo[tipo];
    if (!ed) return;

    TIPO_ATUAL     = tipo;
    EDICAO_ID      = ed.id;
    TIPO_EDICAO_ID = ed.tipo_edicao_id || null;

    // Atualiza nome do tipo na capa
    $('.tipo-atual-nome').text(tipo);

    // Sincroniza o campo de data com a última edição do tipo selecionado
    // para que "Buscar edição" parta da data correta
    $('#dataEdicaoPortal').val(ed.data);

    // Atualiza metadados da edição
    $('.numero-edicao').html(formatar(ed.numero));
    $('.data-edicao').html(formatarData(ed.data));
    $('.paginas-edicao').html(ed.paginas ? ' · ' + ed.paginas + ' págs.' : '');
    $('#input-tipo').val(ed.tipo_edicao_id);

    // Atualiza capa e links de download
    carregarImagem(EDICAO_ID);
    definirUrlDiarios(EDICAO_ID);

    // Sobe a página até a capa
    $('html, body').animate({ scrollTop: 0 }, 400);
}

function carregarCapasTipos(slugs) {
    if (!slugs || slugs.length === 0) return;
    slugs.forEach(function(slug) {
        $.ajax({
            url: BASE_URL + '/apifront/portal/edicoes/edicoes_from_data.json',
            data: { subtheme: slug }
        }).done(function(data) {
            if (data.erro || !data.itens || !data.itens[0]) return;
            var edicao = data.itens[0];
            var edicaoId = edicao.id;
            var imgUrl = (CDN_URL !== '' ? CDN_URL : BASE_URL) +
                         '/' + slug +
                         '/apifront/portal/edicoes/imagem_diario/' + edicaoId + '/1/imagem';
            var $img = $('#capa-tipo-' + slug);
            var $loading = $('#loading-tipo-' + slug);
            var im = new Image();
            im.onload = function() {
                $img.attr('src', imgUrl).fadeIn('fast');
                $loading.hide();
            };
            im.onerror = function() {
                $loading.hide();
            };
            im.src = imgUrl;
            $('.numero-edicao-' + slug).html(formatar(edicao.numero));
            $('.data-edicao-' + slug).html(edicao.data);
            var urlPdf = BASE_URL + '/' + slug + '/portal/visualizacoes/pdf/' + edicaoId + '/#e:' + edicaoId;
            $('#btn-pdf-tipo-' + slug).attr('href', urlPdf).show();
        });
    });
}

function formatarData(dataStr) {
    var meses = ['janeiro','fevereiro','março','abril','maio','junho','julho','agosto','setembro','outubro','novembro','dezembro'];
    var p = (dataStr || '').split('/');
    if (p.length !== 3) return dataStr;
    var dia = parseInt(p[0], 10);
    var mes = parseInt(p[1], 10) - 1;
    return dia + ' de ' + meses[mes] + ' de ' + p[2];
}

function formatar(n) {
    var x1 = n + '';
    var rgx = /(\d+)(\d{3})/;
    while (rgx.test(x1)) {
        x1 = x1.replace(rgx, '$1' + '.' + '$2');
    }
    return x1;
}

function from_data(dataEd) {
    var url = BASE_URL + ((SUBTEMA) ? '/' + SUBTEMA : '') + '/apifront/portal/edicoes/edicoes_from_data/' + dataEd + '.json';
    if (TIPO_EDICAO_ID) {
        url += '?tipoEdicao=' + TIPO_EDICAO_ID;
    }
    $.ajax({
        url: url,
    }).done(function(obj) {
        if (obj.erro) {
            alert(obj.msg);
            return;
        }
        limpaAbas();

        if (dataEd.length > 0) {
          carregaEdicao(obj);
        } else {
          $("#selectEdicaoPortal").html("");
        }
    });
}

function from_numero(numero) {
    var url = BASE_URL + ((SUBTEMA) ? '/' + SUBTEMA : '') + '/apifront/portal/edicoes/edicoes_from_data/' + numero + '.json?buscaPor=numero';
    if (TIPO_EDICAO_ID) {
        url += '&tipoEdicao=' + TIPO_EDICAO_ID;
    }
    $.ajax({
        url: url,
    }).done(function(obj) {
        if (obj.erro) {
            alert(obj.msg);
            return;
        }
        limpaAbas();

        if (numero.length > 0) {
          carregaEdicao(obj);
        } else {
          $("#selectEdicaoPortal").html("");
        }
    });
}

function carregaEdicao(obj) {
  var op = "";

  if (obj.itens && obj.itens[0]) {
      // Prioriza o tipo atualmente selecionado, senão usa o primeiro
      var itemSelecionado = obj.itens[0];
      if (TIPO_ATUAL) {
          for (var t in obj.itens) {
              if (obj.itens[t].tipo_edicao_nome === TIPO_ATUAL) {
                  itemSelecionado = obj.itens[t];
                  break;
              }
          }
      }

      for (var t in obj.itens) {
          op += '<option value="' + obj.itens[t].id + '" ' + ((itemSelecionado.id == obj.itens[t].id) ? 'selected="selected"' : '') + '>' + obj.itens[t].tipo_edicao_nome + '</option>'
      }
      EDICAO_ID  = itemSelecionado.id;
      TIPO_ATUAL = itemSelecionado.tipo_edicao_nome || TIPO_ATUAL;
      carregarImagem(EDICAO_ID);
      populaAbas(obj.itens);
      $('.tipo-atual-nome').text(TIPO_ATUAL);
      $('.numero-edicao').html(formatar(itemSelecionado.numero));
      $('.data-edicao').html(formatarData(itemSelecionado.data));
      $('#dataEdicaoPortal').val(itemSelecionado.data);
      $('.paginas-edicao').html(itemSelecionado.paginas ? ' · ' + itemSelecionado.paginas + ' págs.' : '');
  } else {
      var i = $('.busca_data');
      i.find('.btn').after('<span style="color: #ffc; font-size: 14px; background: #900; padding: 6px; float: left;" class="erro">Não há diário para esta data.</span>');
  }

  $("#selectEdicaoPortal").html(op);
}

function changeFace() {
    var numeroEd = $('#numeroEdicao').val().trim();
    if (numeroEd.length > 0) {
      return from_numero(numeroEd);
    }

    var datat = $('#dataEdicaoPortal').val().split('/');
    dataf = datat[2] + '-' + datat[1] + '-' + datat[0];

    from_data(dataf);
    var i = $('.busca_data');
    i.find('.erro').remove();
    return false;
}

function mudarLabelEdicao(edicao_id) {
    $.ajax({
        url: BASE_URL + '/apifront/portal/edicoes/busca_edicao/' + edicao_id,
        data: {
            subtheme: SUBTEMA
        }
    }).done(function(data) {
        var obj = jQuery.parseJSON(data);
        if (obj.erro) {
            alert(obj.msg);
            return;
        }

        $('.numero-edicao').html(formatar(obj.Edicao.numero));
        $('.data-edicao').html(formatarData(obj.Edicao.data));
        $('.paginas-edicao').html(obj.Edicao.paginas ? ' · ' + obj.Edicao.paginas + ' págs.' : '');
    });
}
