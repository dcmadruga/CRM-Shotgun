(function () {
    var overlay   = document.getElementById('modalProtocolo');
    var btnFechar = document.getElementById('btn-fechar-protocolo');
    var scrollY   = 0;

    function abrirModal() {
        scrollY = window.scrollY || window.pageYOffset;
        overlay.classList.add('aberto');
        document.documentElement.style.overflow = 'hidden';
        document.body.style.overflow = 'hidden';
    }

    function fecharModal() {
        overlay.classList.remove('aberto');
        document.documentElement.style.overflow = '';
        document.body.style.overflow = '';
        window.scrollTo(0, scrollY);
    }

    document.body.appendChild(overlay);
    btnFechar.addEventListener('click', fecharModal);
    overlay.addEventListener('click', function (e) { if (e.target === overlay) fecharModal(); });
    document.addEventListener('keydown', function (e) { if (e.key === 'Escape') fecharModal(); });

    document.getElementById('input-protocolo').addEventListener('keydown', function (e) {
        var allowed = ['Backspace','Delete','ArrowLeft','ArrowRight','ArrowUp','ArrowDown','Tab','Home','End','Enter'];
        if (allowed.indexOf(e.key) !== -1) return;
        if ((e.ctrlKey || e.metaKey) && ['a','c','v','x'].indexOf(e.key.toLowerCase()) !== -1) return;
        if (!/^[0-9\/]$/.test(e.key)) e.preventDefault();
    });

    document.getElementById('form-busca-protocolo').addEventListener('submit', function (e) {
        e.preventDefault();

        var protocolo = document.getElementById('input-protocolo').value.trim();
        var erroEl    = document.getElementById('protocolo-erro');
        var btnEl     = document.getElementById('btn-buscar-protocolo');

        if (!protocolo) {
            erroEl.textContent = 'Informe o número do protocolo.';
            erroEl.style.display = 'block';
            return;
        }

        erroEl.style.display = 'none';
        btnEl.disabled = true;
        btnEl.innerHTML = '<svg width="18" height="18" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><circle cx="11" cy="11" r="8"/><path d="m21 21-4.35-4.35"/></svg> Buscando...';

        $.ajax({
            type: 'POST',
            url: BASE_URL + '/portal/edicoes/consulta_aut_materia',
            data: { protocolo: protocolo },
            dataType: 'json'
        }).done(function (res) {
            if (res.error) {
                erroEl.textContent = res.message || 'Matéria não localizada.';
                erroEl.style.display = 'block';
                return;
            }

            var materia     = res.data.materia;
            var urlDlDiario = BASE_URL + '/portal/edicoes/download/' + materia.edicao_id;
            var urlDlPagina = BASE_URL + '/portal/edicoes/download/' + materia.edicao_id + '/' + (materia.pagina_inicial || 1);

            var dt = materia.data ? materia.data.split('-') : [];
            var dataFormatada = dt.length === 3 ? dt[2] + '/' + dt[1] + '/' + dt[0] : materia.data;

            var protocolo = materia.protocolo + '';
            protocolo = protocolo.replace(/(.{4})$/, '/$1');

            document.getElementById('proto-info').innerHTML =
                '<strong>Protocolo:</strong> ' + protocolo + '<br>' +
                '<strong>Data de publicação:</strong> ' + dataFormatada;

            document.getElementById('btn-download-diario').href = urlDlDiario;
            document.getElementById('btn-download-pagina').href = urlDlPagina;

            // Thumb da capa da edição
            var thumbEl   = document.getElementById('proto-thumb');
            var loadingEl = document.getElementById('proto-thumb-loading');
            var thumbUrl  = BASE_URL + '/apifront/portal/edicoes/imagem_diario/' + materia.edicao_id + '/1/thumb';
            thumbEl.style.display = 'none';
            loadingEl.style.display = 'flex';
            var im = new Image();
            im.onload = function() {
                thumbEl.src = thumbUrl;
                thumbEl.style.display = 'block';
                loadingEl.style.display = 'none';
            };
            im.onerror = function() { loadingEl.style.display = 'none'; };
            im.src = thumbUrl;

            abrirModal();
        }).fail(function (xhr) {
            var msg = (xhr.responseJSON && xhr.responseJSON.message)
                ? xhr.responseJSON.message
                : 'Não foi possível conectar ao servidor. Tente novamente.';
            erroEl.textContent = msg;
            erroEl.style.display = 'block';
        }).always(function () {
            btnEl.disabled = false;
            btnEl.innerHTML = '<svg width="18" height="18" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><circle cx="11" cy="11" r="8"/><path d="m21 21-4.35-4.35"/></svg> Buscar matéria';
        });
    });
})();

    <script>
    (function () {
        var overlay  = document.getElementById('modalAutenticidade');
        var btnAbrir = document.getElementById('btn-abrir-autenticidade');
        var btnFechar = document.getElementById('btn-fechar-autenticidade');
        var inputHash = document.getElementById('aut-input-hash');
        var btnConsultar = document.getElementById('aut-btn-consultar');
        var resultado = document.getElementById('aut-resultado');

        /* Move o overlay para filho direto do body ao iniciar,
           evitando que transform/backdrop-filter de elementos pai
           quebrem o position:fixed no mobile */
        document.body.appendChild(overlay);

        /* Salva posição do scroll para restaurar ao fechar (fix iOS) */
        var scrollY = 0;

        function abrir() {
            scrollY = window.scrollY || window.pageYOffset;
            overlay.classList.add('aberto');
            /* trava scroll em todos os browsers incluindo iOS Safari */
            document.documentElement.style.overflow = 'hidden';
            document.body.style.overflow = 'hidden';
            inputHash.value = '';
            resultado.style.display = 'none';
            resultado.innerHTML = '';
            setTimeout(function () { inputHash.focus(); }, 100);
        }

        function fechar() {
            overlay.classList.remove('aberto');
            document.documentElement.style.overflow = '';
            document.body.style.overflow = '';
            window.scrollTo(0, scrollY);
        }

        btnAbrir.addEventListener('click', function (e) { e.preventDefault(); abrir(); });
        btnFechar.addEventListener('click', fechar);
        overlay.addEventListener('click', function (e) { if (e.target === overlay) fechar(); });
        document.addEventListener('keydown', function (e) { if (e.key === 'Escape') fechar(); });

        function consultar() {
            var hash = inputHash.value.trim();
            if (!hash) { inputHash.focus(); return; }

            btnConsultar.disabled = true;
            btnConsultar.textContent = 'Consultando...';
            resultado.style.display = 'none';
            resultado.innerHTML = '';

            $.ajax({
                type: 'POST',
                dataType: 'json',
                url: BASE_URL + '/portal/edicoes/consulta_autenticidade',
                data: { hash: hash }
            }).done(function (res) {
                mostrarResultado(res);
            }).fail(function (e) {
                mostrarErro(e.responseJSON.message);
            }).always(function () {
                btnConsultar.disabled = false;
                btnConsultar.textContent = 'Consultar';
            });
        }

        function mostrarResultado(res) {
            resultado.style.display = 'block';

            if (res.error) {
                mostrarErro(res.msg || 'Código não encontrado.');
                return;
            }

            var edicaoId = res.data.Edicao && res.data.Edicao.id ? res.data.Edicao.id : null;
            var urlVer   = edicaoId
                ? BASE_URL + '/ver-html/' + edicaoId + '/#e:' + edicaoId
                : null;
            var urlPdf   = edicaoId
                ? BASE_URL + '/portal/edicoes/download/' + edicaoId
                : null;

            var html = '<div class="aut-resultado-ok">';
            html += '<div class="aut-resultado-ok-header">';
            html += '<svg width="15" height="15" fill="none" stroke="currentColor" stroke-width="2.5" viewBox="0 0 24 24"><polyline points="20 6 9 17 4 12"/></svg>';
            html += 'Autenticidade confirmada';
            html += '</div>';
            html += '<div class="aut-resultado-ok-body">';
            html += res.msg || '';

            if (urlPdf) {
                html += '<a href="' + urlPdf + '" class="aut-btn-ver">';
                html += '<svg width="13" height="13" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><path d="M14 2H6a2 2 0 0 0-2 2v16a2 2 0 0 0 2 2h12a2 2 0 0 0 2-2V8z"/><polyline points="14 2 14 8 20 8"/></svg>';
                html += 'PDF</a>&nbsp;';
            }

            if (urlVer) {
                html += '<a href="' + urlVer + '" class="aut-btn-ver">';
                html += '<svg width="13" height="13" fill="none" stroke="currentColor" stroke-width="2.5" viewBox="0 0 24 24"><circle cx="11" cy="11" r="8"/><path d="m21 21-4.35-4.35"/></svg>';
                html += 'Visualizar edição</a>';
            }
            html += '</div></div>';

            resultado.innerHTML = html;
        }

        function mostrarErro(msg) {
            resultado.style.display = 'block';
            resultado.innerHTML = '<div class="aut-resultado-erro">'
                + '<svg width="16" height="16" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24"><circle cx="12" cy="12" r="10"/><line x1="12" y1="8" x2="12" y2="12"/><line x1="12" y1="16" x2="12.01" y2="16"/></svg>'
                + '<span>' + msg + '</span>'
                + '</div>';
        }

        btnConsultar.addEventListener('click', consultar);
        inputHash.addEventListener('keydown', function (e) { if (e.key === 'Enter') consultar(); });

        var params = new URLSearchParams(window.location.search);
        var hash = params.get('hash');

        if (hash != null && hash.length > 0) {
            abrir();
            $("#aut-input-hash").val(hash);
            consultar();
        }

    })();
    </script>
